using System.Threading.Tasks;
using Own.Protocols.Sample;

namespace Own.Protocols.Cases;

// A token cannot survive an await: the callback cannot be async.
public static class C5AsyncRegion
{
    public static async Task Run(Db db, int id)
    {
        var order = await db.SingleAsync(id);
        Protocol.WithApproved(order, async approved =>
        {
            await db.SaveChangesAsync();
            approved.Ship();
        });
        await db.SaveChangesAsync();
    }
}
