using System.Threading.Tasks;
using Own.Protocols.Sample;

namespace Own.Protocols.Cases;

// N1: a token cannot be carried into ANOTHER region - a ref struct cannot be captured.
public static class C2CaptureToken
{
    public static async Task Run(Db db, int id)
    {
        var order = await db.SingleAsync(id);
        var other = await db.SingleAsync(id + 1);
        Protocol.WithApproved(order, approved =>
        {
            Protocol.WithDraft(other, draft =>
            {
                approved.Ship();     // a token of `order` inside the region of `other`
                draft.Cancel();
            });
        });
        await db.SaveChangesAsync();
    }
}
