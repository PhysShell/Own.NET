using System;
using System.Threading;
using System.Threading.Tasks;
using Microsoft.AspNetCore.Http;
using Microsoft.EntityFrameworkCore;
using OrderBackend.Data;
using OrderBackend.Domain;

namespace OrderBackend.Refused;

// The domain sits in the SAME project as the handlers, so the token's `internal` constructor
// is reachable: this ships an order whose state nobody checked, in a handler with no region.
// The compiler accepts it; the scan must not.
public static class ForgedToken
{
    public static async Task<IResult> Ship(int id, AppDbContext db, CancellationToken ct)
    {
        var order = await db.Orders.SingleAsync(x => x.Id == id, ct);
        var now = DateTime.UtcNow;
        new ApprovedOrder(order).Ship(now);
        await db.SaveChangesAsync(ct);
        return Results.Ok();
    }
}
