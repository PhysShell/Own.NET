using System;
using System.IO;
using System.Threading.Tasks;
using Own.Protocols.Sample;

namespace Own.Protocols.Cases;

// A call inside the region that never mentions the entity, and mutates it anyway.
// Nothing at the call site touches the entity, so since OwnIR v2 the lowering hands the call
// to the core as a `proven_call` - and the core REFUSES it: the heap-effect summaries see the
// mutation behind it (AnnotateLast writes an object reached through a static, writes.instance
// is may). An exclusive region does not analyse unprovable code as safe. REFUSED, never clean
// - one stage later than before H1, for a reason the summaries can now name.
public static class R9OpaqueCallInsideRegion
{
    public static async Task Run(Db db, int id)
    {
        var order = await db.SingleAsync(id);
        Backdoor.Remember(order);
        Protocol.WithApproved(order, approved =>
        {
            Backdoor.AnnotateLast(); // mutates `order`; nothing here says so
            approved.Ship();
        });
        await db.SaveChangesAsync();
    }
}
